cmake_minimum_required(VERSION 3.24)
project(Trickster VERSION 0.5.0 LANGUAGES C CXX)
if(MSVC AND DEFINED TRICKSTER_MSVC_INCLUDE_PREFIX)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${TRICKSTER_MSVC_INCLUDE_PREFIX}")

set(TRICKSTER_BUILD_ID "unknown")
find_package(Git QUIET)
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CMAKE_CURRENT_SOURCE_DIR}" rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE TRICKSTER_BUILD_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CMAKE_CURRENT_SOURCE_DIR}" status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE TRICKSTER_DIRTY ERROR_QUIET)
  if(TRICKSTER_DIRTY)
    string(APPEND TRICKSTER_BUILD_ID "-dirty")
  endif()
endif()
if(NOT TRICKSTER_BUILD_ID)
  set(TRICKSTER_BUILD_ID "unknown")
endif()
configure_file(src/version.h.in generated/trickster_version.h @ONLY)
add_library(ufbx STATIC third_party/ufbx/ufbx.c)

# Official unmodified native SDK; runtime has no Python dependency.
set(TRICKSTER_MUJOCO_VERSION 3.3.7)
if(WIN32)
  include(FetchContent)
  FetchContent_Declare(mujoco_sdk
    URL https://github.com/google-deepmind/mujoco/releases/download/3.3.7/mujoco-3.3.7-windows-x86_64.zip
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    URL_HASH SHA256=1108ad01f6a1e63d42725188404c9af50b81902c368ef026d20d5148c115a08b)
  FetchContent_MakeAvailable(mujoco_sdk)
  add_library(mujoco::mujoco SHARED IMPORTED GLOBAL)
  set_target_properties(mujoco::mujoco PROPERTIES
    IMPORTED_IMPLIB "${mujoco_sdk_SOURCE_DIR}/lib/mujoco.lib"
    IMPORTED_LOCATION "${mujoco_sdk_SOURCE_DIR}/bin/mujoco.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${mujoco_sdk_SOURCE_DIR}/include")
  configure_file("${mujoco_sdk_SOURCE_DIR}/bin/mujoco.dll" "${CMAKE_BINARY_DIR}/mujoco.dll" COPYONLY)
else()
  find_package(mujoco 3.3.7 EXACT REQUIRED CONFIG)
endif()
set_target_properties(ufbx PROPERTIES AUTOMOC OFF AUTOUIC OFF)
add_library(TricksterCore src/notes_navigation.cpp src/motion_expectations.cpp src/curve_bake.cpp src/motion_validation.cpp src/analysis_jobs.cpp src/motion_compare.cpp src/retarget.cpp src/motion_analysis.cpp src/animation_curves.cpp src/motion_annotations.cpp src/physics_proxy.cpp src/physics_trajectory.cpp src/physics_validation.cpp src/physics_demos.cpp src/demo_scenes.cpp src/recovery.cpp src/evaluator.cpp src/key_editing.cpp src/edit_service.cpp src/physics_jobs.cpp src/workspace.cpp src/profile.cpp src/motion.cpp src/physics.cpp src/mujoco_backend.cpp src/whole_body.cpp src/flight.cpp src/dynamics_optimizer.cpp src/physics_settings.cpp src/physics_telemetry.cpp src/physics_provenance.cpp src/physics_replay.cpp src/fbx.cpp src/fbx_export.cpp src/core.cpp src/geometry.cpp src/posing.cpp src/gltf.cpp src/document.cpp src/scene_catalog.cpp)

if(WIN32)
 target_link_libraries(TricksterCore PRIVATE Psapi)

add_library(TricksterUI src/notes_card.cpp src/notes_ui.cpp src/notes_overlay.cpp src/analysis_views.cpp src/retarget_ui.cpp src/motion_analysis_ui.cpp src/graph_editor.cpp src/physics_ui.cpp src/physics_job.cpp src/profile_ui.cpp src/motion_ui.cpp src/icons.cpp src/panels_ui.cpp src/layout_ui.cpp src/window.cpp src/helpers_ui.cpp src/scenes_ui.cpp src/viewport.cpp src/timeline.cpp src/editor_api.cpp src/editor_ui_api.cpp src/mcp_server.cpp src/automation_ui.cpp)
